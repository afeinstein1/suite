
add_library(platform_core STATIC
    src/platform.cpp
)

target_include_directories(platform_core
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_compile_features(platform_core PUBLIC cxx_std_20)

add_executable(platform_app
    src/main.cpp
)

target_link_libraries(platform_app
    PRIVATE platform_core
)
